### 1. 两数之和
```
1. 给定一个整数数组 nums 和一个整数目标值 target，请你在该数组中找出 和为目标值 target  的那 两个 整数，并返回它们的数组下标。
2. 你可以假设每种输入只会对应一个答案。但是，数组中同一个元素在答案里不能重复出现。
3. 你可以按任意顺序返回答案。
4. 示例：
    1. 输入：nums = [2,7,11,15], target = 9
    2. 输出：[0,1]
    3. 解释：因为 nums[0] + nums[1] == 9 ，返回 [0, 1] 。
```
```javaScript
// 哈希 target - nums[i] 取到了则返回
function twoSum(nums, target) {
    const map = new Map();
    for (let i = 0; i < nums.length; i++) {
        const diff = target - nums[i];
        if (map.has(diff)) {
            return [map.get(diff), i];
        }
        map.set(nums[i], i);
    }
    return [];
}
```

```javaScript
function groupAnagrams(strs) {
    const map = new Map();

        const key = strs[i].split('').sort().join('');
        if (map.has(key)) {

        }
```

### 3. 最长连续序列
```
1. 给定一个未排序的整数数组 nums ，找出数字连续的最长序列（不要求序列元素在原数组中连续）的长度。
2. 请你设计并实现时间复杂度为 O(n) 的算法解决此问题。
3. 示例：
    1. 输入：nums = [100,4,200,1,3,2]
    2. 输出：4
    3. 解释：最长数字连续序列是 [1, 2, 3, 4]。它的长度为 4。
```
```javaScript
// 哈希表 找到前一个的话就跳过，确保是从连续的第一个开始
var longestConsecutive = function(nums) {
    let ans = 0;
    const numSet = new Set(nums);
    numSet.forEach(num => {
        if (numSet.has(num - 1)) {
            return;
        }
        let tempAns = 0;
        let tempNum = num;
        while(numSet.has(tempNum)) {
            tempAns++;
            tempNum++;
        };
        ans = Math.max(tempAns, ans);
    })
    return ans;
};
```
### 4. 移动零
```
1. 给定一个数组 nums，编写一个函数将所有 0 移动到数组的末尾，同时保持非零元素的相对顺序。
2. 请注意 ，必须在不复制数组的情况下原地对数组进行操作。
3. 示例：
    1. 输入: nums = [0,1,0,3,12]
    2. 输出: [1,3,12,0,0]
```
```javaScript
// 双指针 左指针指向0 右指针指向非0
// 关键点： 指向 0 的指针只有在交换时才会移动
function moveZeroes(nums) {
    let left = 0;
    let right = 0;
    while (right < nums.length) {
        if (nums[right] !== 0) {
            [nums[left], nums[right]] = [nums[right], nums[left]];
            left++;
        }
        right++;
    }
    while(left < nums.length) {
        nums[left] = 0;
        left++;
    }
    return nums;
}
```
### 5. 盛最多水的容器
```

2. 找出其中的两条线，使得它们与 x 轴共同构成的容器可以容纳最多的水。
3. 返回容器可以储存的最大水量。
4. 说明：你不能倾斜容器。
5. 示例：
    1. 输入：[1,8,6,2,5,4,8,3,7]
    2. 输出：49 
    3. 解释：图中垂直线代表输入数组 [1,8,6,2,5,4,8,3,7]。在此情况下，容器能够容纳水（表示为蓝色部分）的最大值为 49。
```
```javaScript
// 双指针 指向高度小的那个指针向中间移动
var maxArea = function(height) {
    let pre = 0;
    let right = height.length - 1;
    let ans = -Infinity;
    while(pre < right) {
        ans = Math.max(ans, (right - pre) * Math.min(height[pre], height[right]));
        if (height[pre] < height[right]) {
            pre++;
        } else {
            right--;
        }
    }
    return ans
};
```

### 6. 三数之和
```
1. 给你一个整数数组 nums ，判断是否存在三元组 [nums[i], nums[j], nums[k]] 满足 i != j、i != k 且 j != k ，同时还满足 nums[i] + nums[j] + nums[k] == 0 。请
2. 你返回所有和为 0 且不重复的三元组。
3. 注意：答案中不可以包含重复的三元组。
4. 示例：
    1. 输入：nums = [-1,0,1,2,-1,-4]
    2. 输出：[[-1,-1,2],[-1,0,1]]
    3. 解释：
        1. nums[0] + nums[1] + nums[2] = (-1) + 0 + 1 = 0 。
        2. nums[1] + nums[2] + nums[4] = 0 + 1 + (-1) = 0 。
        3. 不同的三元组是 [-1,0,1] 和 [-1,-1,2] 。
        4. 注意，输出的顺序和三元组的顺序并不重要。
```
```javaScript
// 排序 + 双指针
var threeSum = function(nums) {
    nums.sort((a, b) => a - b);
    const ans = [];
    for (let i = 0; i < nums.length; i++) {
        if (nums[i] > 0) {
            return ans;
        }
        if (i > 0 && nums[i] === nums[i - 1]) {
            continue;
        }
        let left = i + 1;
        let right = nums.length - 1;
        while (left < right) {
            const sum = nums[i] + nums[left] + nums[right];
            if (sum > 0) {
                right--;
            } else if (sum < 0) {
                left++;
            } else {
                ans.push([nums[i], nums[left], nums[right]]);
                while (left < right && nums[left] === nums[left + 1]) {
                    // 去重
                    left++;
                }
                while (left < right && nums[right] === nums[right - 1]) {
                    // 去重
                    right--;
                }
                left++;
                right--;
            }
        }
    }
    return ans;
};
```

### 7. 接雨水
```
1. 给定 n 个非负整数表示每个宽度为 1 的柱子的高度图，计算按此排列的柱子，下雨之后能接多少雨水。
2. 示例：
    1. 输入：height = [0,1,0,2,1,0,1,3,2,1,2,1]
    2. 输出：6
    3. 解释：上面是由数组 [0,1,0,2,1,0,1,3,2,1,2,1] 表示的高度图，在这种情况下，可以接 6 个单位的雨水（蓝色部分表示雨水）。 
```
```javaScript
// 双指针
var trap = function(height) {
    let left = 0;
    let right = height.length - 1;
    let ans = 0;
    let preMax = 0;
    let sufMax = 0;
    while (left < right) {
        preMax = Math.max(preMax, height[left]);
        sufMax = Math.max(sufMax, height[right]);
        if (height[left] < height[right]) {
            ans += preMax - height[left];
            left++;
        } else {
            ans += sufMax - height[right];
            right--;
        }
    }
    return ans;
};

var trap = function(height) {
    // 单调栈
    const stack = [];
    let ans = 0;
    for (let i = 0; i < height.length; i++) {
        while (stack.length && height[stack[stack.length - 1]] < height[i]) {
            const mid = stack.pop();
            if (stack.length) {
                ans += (i - stack[stack.length - 1] - 1) * (Math.min(height[i], height[stack[stack.length - 1]]) - height[mid]);
            }
        }
        stack.push(i);
    }
    return ans;
}
```

### 8. 无重复字符的最长子串
```
1. 给定一个字符串 s ，请你找出其中不含有重复字符的 最长子串 的长度。
2. 示例：
    1. 输入: s = "abcabcbb"
    2. 输出: 3 
    3. 解释: 因为无重复字符的最长子串是 "abc"，所以其长度为 3。
```
```javaScript
// 滑动窗口
function lengthOfLongestSubstring(s) {
    const map = new Map();
    let left = 0;
    let maxLength = 0;
    for (let right = 0; right < s.length; right++) {
        if (map.has(s[right])) {
            left = Math.max(left, map.get(s[right]) + 1);
        }
        map.set(s[right], right);
        maxLength = Math.max(maxLength, right - left + 1);
    }
    return maxLength;
}
```

```javaScript
const findAnagrams = (s, p) => {
```

### 10. 和为 K 的子数组

```javaScript
// 前缀和 + 哈希表
var subarraySum = function(nums, k) {
    const map = new Map();
    map.set(0, 1);
    let preSum = 0;
    let count = 0;
    for (let i = 0; i < nums.length; i++) {
        preSum += nums[i];
        if (map.has(preSum - k)) {
            count += map.get(preSum - k);
        }
        map.set(preSum, (map.get(preSum) || 0) + 1);
    }
    return count;
};
```

```javaScript
// 排序
const merge = (intervals) => {
    intervals.sort((a, b) => a[0] - b[0]);
    const res = [];
    let pre = intervals[0];
    for (let i = 1; i < intervals.length; i++) {
        if (pre[1] >= intervals[i][0]) {
            pre[1] = Math.max(pre[1], intervals[i][1]);
        } else {
            res.push(pre);
            pre = intervals[i];
        }
    }
    res.push(pre);
```

### 13. 旋转数组
```

2. 示例：
    1. 输入: nums = [1,2,3,4,5,6,7], k = 3

    3. 解释:
        1. 向右旋转 1 步: [7,1,2,3,4,5,6]
```

```javaScript
// 旋转数组
// 分3次反转
const rotate = (nums, k) => {
    const swapArr = (arr) => {
        const tempArr = arr.slice();
        let left = 0, right = tempArr.length - 1;
        while (left < right) {
            [tempArr[left], tempArr[right]] = [tempArr[right], tempArr[left]];
            left++;
            right--;
        }
        return tempArr;
    }
    k = k % nums.length;
    nums.splice(0, nums.length, ...swapArr(nums));
    nums.splice(0, k, ...swapArr(nums.slice(0, k)));
    nums.splice(k, nums.length, ...swapArr(nums.slice(k)));
    return nums;
}
```

```javaScript
// 除自身以外数组的乘积
const productExceptSelf = (nums) => {
    const res = new Array(nums.length).fill(1);
    let pre = 1;
    for (let i = 0; i < nums.length; i++) {
        res[i] = pre;
```

### 15. 缺失的第一个正数
```
1. 给你一个未排序的整数数组 nums ，请你找出其中没有出现的最小的正整数。
2. 示例：
    1. 输入：nums = [1,2,0]
    2. 输出：3
```
```javaScript
// 缺失的第一个正数
// 最关键的还是将每个数字放到正确的位置上，看 while 判断条件
var firstMissingPositive = function(nums) {
    const n = nums.length;
    for (let i = 0; i < n; i++) {
        // 如果当前学生的学号在 [1,n] 中，但（真身）没有坐在正确的座位上
        while (1 <= nums[i] && nums[i] <= n && nums[i] !== nums[nums[i] - 1]) {
            // 那么就交换 nums[i] 和 nums[j]，其中 j 是 i 的学号
            const j = nums[i] - 1; // 减一是因为数组下标从 0 开始
            [nums[i], nums[j]] = [nums[j], nums[i]];
        }
    }

    // 找第一个学号与座位编号不匹配的学生
    for (let i = 0; i < n; i++) {
        if (nums[i] !== i + 1) {
            return i + 1;
        }
    }

    // 所有学生都坐在正确的座位上
    return n + 1;
};
```

```javaScript
const setZeroes = (matrix) => {
    const m = matrix.length, n = matrix[0].length;
    const row = new Array(m).fill(false);
```

### 17. 螺旋矩阵
```
1. 给你一个 m 行 n 列的矩阵 matrix ，请按照 顺时针螺旋顺序 ，返回矩阵中的所有元素。
2. 示例：

    2. 输出：[1,2,3,6,9,8,7,4,5]
```
```javaScript
// 螺旋矩阵
const spiralOrder = (matrix) => {
    const res = [];
    const m = matrix.length, n = matrix[0].length;
    let top = 0, bottom = m - 1, left = 0, right = n - 1;
    while (top <= bottom && left <= right) {
        // 从左到右
        for (let i = left; i <= right; i++) {
            res.push(matrix[top][i]);
        }
        top++;
        // 从上到下
        for (let i = top; i <= bottom; i++) {
            res.push(matrix[i][right]);
        }
        right--;
        // 从右到左
        for (let i = right; i >= left; i--) {
            res.push(matrix[bottom][i]);
        }
        bottom--;
        // 从下到上
        for (let i = bottom; i >= top; i--) {
            res.push(matrix[i][left]);
        }
        left++;
    }
    return res;
}
```

### 19. 搜索二维矩阵

```javaScript
// 搜索二维矩阵
const searchMatrix = (matrix, target) => {
    const m = matrix.length, n = matrix[0].length;
    let i = 0, j = n - 1;
    while (i < m && j >= 0) {
        if (matrix[i][j] === target) {
            return true;
        } else if (matrix[i][j] > target) {
            j--;
        } else {
            i++;
        }
    }
```
